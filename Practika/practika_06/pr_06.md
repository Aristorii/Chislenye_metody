# Задание 1
```python
import numpy as np
import matplotlib.pyplot as plt

# 1. Построение правильного шестиугольника
k = np.arange(6)
x = np.cos(2 * np.pi * k / 6)
y = np.sin(2 * np.pi * k / 6)

# Замыкаем контур (добавляем первую вершину в конец)
hexagon = np.column_stack((x, y))
hexagon_closed = np.vstack([hexagon, hexagon[0]])

# 2. Матрица поворота на 45 градусов (против часовой)
theta = np.deg2rad(45)
rotate_matrix = np.array([
    [np.cos(theta), -np.sin(theta)],
    [np.sin(theta),  np.cos(theta)]
])

# Применяем поворот (умножаем матрицу на транспонированные точки)
rotated = (rotate_matrix @ hexagon_closed.T).T

# 3. Матрица отражения относительно прямой y = x
reflect_yx = np.array([
    [0, 1],
    [1, 0]
])

# Применяем отражение к исходному шестиугольнику
reflected = (reflect_yx @ hexagon_closed.T).T

# 4. Визуализация
fig, ax = plt.subplots(figsize=(6, 6))

# Исходный (серый)
ax.plot(hexagon_closed[:, 0], hexagon_closed[:, 1], 'o-', color='gray', label='Исходный')
# Повернутый (синий)
ax.plot(rotated[:, 0], rotated[:, 1], 'o-', color='steelblue', label='Поворот на 45°')
# Отраженный (красный)
ax.plot(reflected[:, 0], reflected[:, 1], 'o-', color='crimson', label='Отражение y=x')

# Отрисовка осей и настройка графика
ax.axhline(0, color='black', lw=0.5)
ax.axvline(0, color='black', lw=0.5)
ax.set_aspect('equal')  # Чтобы круг был кругом, а квадрат квадратом
ax.grid(True, alpha=0.3)
ax.legend()
ax.set_title('Поворот и отражение шестиугольника')
plt.show()

# --- ОТВЕТ НА ВОПРОС ---
# Отраженный шестиугольник выглядит "повернутым", потому что отражение 
# относительно прямой y = x математически эквивалентно транспонированию 
# координат (x, y) -> (y, x). Для правильного многоугольника это визуально 
# совпадает с поворотом на 90 градусов (матрицы поворота на 90° и 
# отражения относительно y=x совпадают с точностью до знака).
```

# Задание 2

```python
import numpy as np
import matplotlib.pyplot as plt

# 1. Генерация 200 случайных точек в квадрате [-2, 2] x [-2, 2]
rng = np.random.default_rng(42)
points = rng.uniform(-2, 2, size=(200, 2))

# Переводим точки в однородные координаты (добавляем единичный столбец Z)
# Это нужно для выполнения переноса (сдвига) через умножение матриц.
points_h = np.hstack([points, np.ones((200, 1))])

# 2. Создание матриц преобразования 3x3
# Масштабирование
S = np.array([
    [1.5, 0.0, 0.0],
    [0.0, 0.5, 0.0],
    [0.0, 0.0, 1.0]
])

# Поворот на 60 градусов
theta = np.deg2rad(60)
R = np.array([
    [np.cos(theta), -np.sin(theta), 0.0],
    [np.sin(theta),  np.cos(theta), 0.0],
    [0.0,            0.0,           1.0]
])

# Перенос на вектор (1, -1)
T = np.array([
    [1.0, 0.0, 1.0],
    [0.0, 1.0, -1.0],
    [0.0, 0.0, 1.0]
])

# 3. Композиция преобразований (Сначала масштаб, потом поворот, потом перенос)
# Порядок умножения: T @ R @ S (справа налево)
M = T @ R @ S

# Применяем преобразование к точкам
transformed_h = (M @ points_h.T).T

# 4. Обратное преобразование
# Обратная матрица M_inv
M_inv = np.linalg.inv(M)

# Восстанавливаем точки
restored_h = (M_inv @ transformed_h.T).T

# Отрезаем однородную координату (оставляем только x, y)
transformed = transformed_h[:, :2]
restored = restored_h[:, :2]

# 5. Проверка точности
is_close = np.allclose(points, restored, atol=1e-12)
print(f"Точки восстановлены с точностью 1e-12: {is_close}")

# 6. Визуализация
fig, ax = plt.subplots(figsize=(8, 8))

ax.scatter(points[:, 0], points[:, 1], color='gray', alpha=0.5, label='Исходные точки')
ax.scatter(transformed[:, 0], transformed[:, 1], color='steelblue', alpha=0.5, label='Преобразованные')
ax.scatter(restored[:, 0], restored[:, 1], color='crimson', marker='x', alpha=0.8, label='Восстановленные')

ax.axhline(0, color='black', lw=0.5)
ax.axvline(0, color='black', lw=0.5)
ax.set_aspect('equal')
ax.grid(True, alpha=0.3)
ax.legend()
ax.set_title('Композиция и восстановление точек')
plt.show()

# --- ОТВЕТ НА ВОПРОС ---
# Обратные преобразования нужно применять в обратном порядке, потому что 
# матричное умножение некоммутативно (A @ B != B @ A). По свойству обратной 
# матрицы: (A @ B)^-1 = B^-1 @ A^-1. Если применить их в прямом порядке, 
# результат будет неверным (точки не совпадут).
```
